#!/usr/bin/env bash
# 在 AIS 的独立作业中，各复制对应的一行命令（从 bash 开始）。
# SFT 续评，8 个 rank，每个 rank 一张卡
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt sft
# Frontier 补评，8 个 rank，每个 rank 一张卡，从 23493157 继承已核验的试次
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt frontier
# 数据筛选，单作业连续筛 16 + 32 题，每题 4 次。8 个单卡 rank 各复用一个冻结的 SFT 服务和两个并发桌面。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen
# 正式过夜筛选，从扩大后的候选池选 256 条。8 个单卡 rank，每题先测初态，再采 4 次。
# rank 0 每 5 分钟导出已评分题和严格复核题；--plan 可查看冻结队列。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-overnight
# 下一批筛选：8 道已通过本地判分检查的新变体，每个 rank 一题，初态零分后各采 4 次。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-next
# 扩围筛选，16 道新的 Calc、Impress 和文件操作题，8 个单卡 rank 各筛两题。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-wide
# 正式扩围，八卡动态领题，每卡两个桌面。各类严核满额即停发；--plan 可查看规模。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-expand
# 主练扩围。八卡共 16 个桌面，新增 133 道候选，每道先测初态，再采 4 次。自动复用前一作业的有效评分。
# 45 道已有四次评分的主练候选只进 CPU 复核清单。作业启动时固定队列，--plan 查看本次规模。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-followup
# 全池连续筛选：接在 23593010 后，同一个八卡作业复核、补缺分、继续领完训练划分的剩余候选，满额即停。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-after 23593010
# 复核使用 CPU。传入正在跑的 screen-expand 作业号，会等待该作业收尾，再核验本轮和历史评分。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt screen-review 23560412
# RL 只提交一个作业。支持 8 个单卡 rank，或 1 个八卡实例。所有 rank 使用同一条 Entry。
# 八个单卡 rank 各启动一个 Ray 节点，默认每个 worker 同时运行 2 个桌面，共 16 路。
# 可在同一条命令前设置 CUA_RL_DESKTOPS_PER_RANK=4 做 32 路对照。每个 prompt 仍采样 8 次。
# bash /home/work/aigc_video_research/wangjing/CODE/CUA-RL/AIS_ENTRY.txt rl
# 将 RL 设为 Frontier 成功后的下游作业。RL 默认使用此次补评核验的成功率。
# 默认留空 CUA_CONTROLLER_URL。只有复用已有的专用服务时才填写外部地址。
# 可用 CUA_FRONTIER_RATES 指定其他成功率文件。
# 八个单卡 rank 可用 rl --check-only 单独验证 NCCL、八个 Ray GPU 和 2+6 放置。正式 rl 会自动执行此检查。
# check 验证训练环境和通信，check --eight-gpus 用于单实例八卡。
set -eu
project=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
entry="$project/scripts/ais_entry.sh"
frontier_source=23493157
frontier_rates="$project/runs/ais-frontier-campaigns/repair-$frontier_source-rates.json"
mode=${1:-check}
if (( $# )); then shift; fi

case "$mode" in
    screen-followup)
        exec python3 -u "$project/scripts/ais_screen_queue.py" --followup "$@"
        ;;
    screen-after)
        exec python3 -u "$project/scripts/ais_screen_queue.py" --followup --after-review "$@"
        ;;
    screen-review)
        exec python3 -u "$project/scripts/review_screen_job.py" "$@"
        ;;
    screen-expand)
        exec python3 -u "$project/scripts/ais_screen_queue.py" "$@"
        ;;
    screen-wide)
        exec python3 -u "$project/scripts/ais_dataset_screen.py" --indices "$project/configs/dataset-screen/wide-20261126.json" --check-initial-state "$@"
        ;;
    screen-next)
        exec python3 -u "$project/scripts/ais_dataset_screen.py" --indices "$project/configs/dataset-screen/expansion-20261126.json" --check-initial-state "$@"
        ;;
    screen-overnight)
        exec python3 -u "$project/scripts/ais_dataset_screen.py" --overnight "$@"
        ;;
    screen)
        exec python3 -u "$project/scripts/ais_dataset_screen.py" "$@"
        ;;
    sft)
        export CUA_SFT_DESKTOPS=2 CUA_SFT_BOOT_MODE=snapshot-gui
        exec "$entry" sft-audit --full --resume-job 23495234 "$@"
        ;;
    frontier)
        if [[ "${1:-}" == --plan ]]; then
            shift
            exec "$entry" frontier --plan-reuse "$frontier_source" "$@"
        fi
        exec "$entry" frontier --reuse-from "$frontier_source" --publish-rates "$frontier_rates" "$@"
        ;;
    rl-desktop)
        worker_python="${CUA_WORKER_PYTHON:-$project/../../ENV/conda_envs/test-gpu/bin/python}"
        export CUA_FRONTIER_WARM_GUEST="${CUA_FRONTIER_WARM_GUEST:-1}"
        exec "$worker_python" -u "$project/scripts/ais_desktop_service.py" --rl-worker "$@"
        ;;
    rl|check)
        export CUA_TRAIN_PYTHON="$project/../../ENV/cua-rl-ais-cu129/bin/ais-python"
        if [[ "$mode" == rl ]]; then
            export CUA_FRONTIER_RATES="${CUA_FRONTIER_RATES:-$frontier_rates}"
            exec "$entry" train "$@"
        fi
        if (( $# == 0 )); then set -- --all-ranks; fi
        exec "$entry" train-check "$@"
        ;;
    *)
        echo 'Usage: bash AIS_ENTRY.txt [sft|frontier|screen|screen-overnight|screen-next|screen-wide|screen-expand|screen-followup|screen-after JOB_ID|screen-review JOB_ID|rl-desktop|rl|check]' >&2
        exit 2
        ;;
esac
